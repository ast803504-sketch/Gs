import socket import requests import json import sys def gather_target_info(target): print(f"[*] جاري اختراق وتحليل الهدف: {target}...") try: ip_address = socket.gethostbyname(target) print(f"[+] عنوان الآبي (IP): {ip_address}") except socket.gaierror: print("[-] فشل في تحليل الآبي يا رئيس.") return api_url = f"http://ip-api.com/json/{ip_address}" response = requests.get(api_url) if response.status_code == 200: data = response.json() print("\n--- [ تفاصيل الموقع والدولة ] ---") print(f"الدولة: {data.get('country', 'N/A')}") print(f"المدينة: {data.get('city', 'N/A')}") print(f"الشركة المزودة (ISP): {data.get('isp', 'N/A')}") print(f"المنظمة: {data.get('org', 'N/A')}") else: print("[-] صار خطأ في جلب بيانات الـ IP يا بوص.") if __name__ == "__main__": if len(sys.argv) < 2: print("[-] الاستخدام الصح: python tool.py example.com") sys.exit(1) target_domain = sys.argv[1] gather_target_info(target_domain)
